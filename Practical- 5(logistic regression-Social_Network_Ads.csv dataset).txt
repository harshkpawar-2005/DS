import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegression

from sklearn.metrics import (
    confusion_matrix,
    accuracy_score,
    precision_score,
    recall_score
)

data = pd.read_csv("Social_Network_Ads.csv")

print("\nSocial Network Ads Dataset")
print(data)

print("\nFirst 5 Rows")
print(data.head())

print("\nMissing Values")
print(data.isnull().sum())

data['Gender'] = data['Gender'].replace({
    'Male': 1,
    'Female': 0
})

print("\nDataset After Encoding")
print(data.head())

print("\nCovariance Matrix")
print(data.cov())

x = data[['Age', 'EstimatedSalary']]

y = data['Purchased']

xtrain, xtest, ytrain, ytest = train_test_split(
    x,
    y,
    test_size=0.25,
    random_state=0
)

sc = StandardScaler()

xtrain = sc.fit_transform(xtrain)

xtest = sc.transform(xtest)

logreg = LogisticRegression()

logreg.fit(xtrain, ytrain)

y_pred = logreg.predict(xtest)

print("\nPredicted Values")
print(y_pred)

cm = confusion_matrix(ytest, y_pred)

print("\nConfusion Matrix")
print(cm)

TN = cm[0][0]

FP = cm[0][1]

FN = cm[1][0]

TP = cm[1][1]

print("\nTrue Positive (TP) =", TP)

print("True Negative (TN) =", TN)

print("False Positive (FP) =", FP)

print("False Negative (FN) =", FN)

accuracy = accuracy_score(ytest, y_pred)

print("\nAccuracy =", accuracy)

error_rate = 1 - accuracy

print("\nError Rate =", error_rate)

precision = precision_score(ytest, y_pred)

print("\nPrecision =", precision)

recall = recall_score(ytest, y_pred)

print("\nRecall =", recall)

plt.scatter(
    xtest[:, 0],
    xtest[:, 1],
    c=y_pred
)

plt.xlabel("Age")

plt.ylabel("Estimated Salary")

plt.title("Logistic Regression Classification")

plt.show()

1. What is Logistic Regression?

Answer:
Logistic Regression is a supervised machine learning algorithm used for classification problems. It predicts categorical output values such as:

- Yes/No
- True/False
- Spam/Not Spam

It is mainly used for binary classification problems where only two output classes are present.



2. Why is Logistic Regression called a classification algorithm?

Answer:
Logistic Regression predicts categories or classes instead of continuous values.

For example:
- Whether a customer will buy a product or not
- Whether a patient has cancer or not

Since the output belongs to predefined classes, it is called a classification algorithm.

5. What is binary classification?

Answer:
Binary classification means the output has only two possible classes.

Examples:
- 0 or 1
- Yes or No
- Spam or Not Spam

Logistic Regression is mainly used for binary classification.



6. What is the difference between Linear Regression and Logistic Regression?

Answer:

Linear Regression
- Used for prediction
- Output is continuous
- Uses straight line equation
- Predicts values like price

Logistic Regression
- Used for classification
- Output is categorical
- Uses sigmoid function
- Predicts classes like Yes/No

Example:
- Linear Regression → House Price Prediction
- Logistic Regression → Disease Prediction



7. What is the sigmoid function?

Answer:
The sigmoid function converts output values into probabilities between 0 and 1.

Formula:

σ(x) = 1 / (1 + e^-x)

It gives an S-shaped curve.

- If output > 0.5 → class 1
- If output < 0.5 → class 0



8. Why is sigmoid function used in Logistic Regression?

Answer:
The sigmoid function is used because classification requires probability output between 0 and 1.

It helps convert linear output into probabilities for decision making.



9. What are the types of Logistic Regression?

Answer:

1. Binary Logistic Regression
- Only two output classes
- Example: Spam/Not Spam

2. Multinomial Logistic Regression
- More than two categories
- Example: Predicting fruit type

3. Ordinal Logistic Regression
- Ordered categories
- Example: Ratings from 1 to 5



10. What is a confusion matrix?

Answer:
A confusion matrix is a table used to evaluate classification model performance.

It compares:
- Actual values
- Predicted values

It helps calculate:
- Accuracy
- Precision
- Recall
- Error Rate



11. What is True Positive (TP)?

Answer:
True Positive means:
- Actual class is positive
- Predicted class is also positive

Example:
A patient has cancer and the model also predicts cancer.



12. What is True Negative (TN)?

Answer:
True Negative means:
- Actual class is negative
- Predicted class is also negative

Example:
A patient does not have cancer and the model predicts no cancer.



13. What is False Positive (FP)?

Answer:
False Positive means:
- Actual class is negative
- Predicted class is positive

It is also called Type-1 Error.

Example:
Model predicts disease but the patient is healthy.



14. What is False Negative (FN)?

Answer:
False Negative means:
- Actual class is positive
- Predicted class is negative

It is also called Type-2 Error.

Example:
Model predicts healthy but the patient actually has disease.



15. What is Accuracy?

Answer:
Accuracy measures how many predictions are correct.

Formula:

Accuracy =
(TP + TN) /
(TP + TN + FP + FN)

Higher accuracy means better model performance.



16. What is Error Rate?

Answer:
Error Rate measures incorrect predictions.

Formula:

Error Rate =
(FP + FN) /
(TP + TN + FP + FN)

Lower error rate indicates better performance.



17. What is Precision?

Answer:
Precision measures how many predicted positive values are actually positive.

Formula:

Precision =
TP / (TP + FP)

High precision means fewer false positives.



18. What is Recall?

Answer:
Recall measures how many actual positive cases are correctly predicted.

Formula:

Recall =
TP / (TP + FN)

High recall means fewer false negatives.



19. Why is feature scaling used?
