import seaborn as sns
import matplotlib.pyplot as plt
import pandas as pd

iris = sns.load_dataset('iris')

print("\n========== IRIS DATASET ==========\n")

print(iris.to_string())

print("\n========== DATASET INFORMATION ==========\n")

print(iris.info())

print("\n========== DATASET SHAPE ==========\n")

print(iris.shape)

print("\n========== MISSING VALUES ==========\n")

print(iris.isnull().sum())

print("\n========== FEATURES AND DATA TYPES ==========\n")

print(iris.dtypes)

numeric_columns = iris.select_dtypes(include=['float64']).columns

for column in numeric_columns:

    plt.figure(figsize=(6,5))

    sns.histplot(
        iris[column],
        bins=20,
        kde=False
    )

    plt.title(f"Histogram of {column}")

    plt.xlabel(column)
    plt.ylabel("Frequency")

    plt.show()

for column in numeric_columns:

    plt.figure(figsize=(6,5))

    sns.boxplot(
        y=iris[column]
    )

    plt.title(f"Boxplot of {column}")

    plt.ylabel(column)

    plt.show()

print("\n========== OBSERVATIONS ==========\n")

print("1. Sepal length and sepal width are approximately normally distributed.")
print("2. Petal length and petal width show clear variation among species.")
print("3. Boxplots help identify outliers in the dataset.")
print("4. Some outliers are present in sepal width.")
print("5. Petal measurements have less overlap compared to sepal measurements.")
