# Import SQLite Library
import sqlite3

# ----------------------------------------------------------
# CONNECT DATABASE
# ----------------------------------------------------------

# Creates or connects to database file
conn = sqlite3.connect("student.db")

# Cursor object is used to execute SQL queries
cursor = conn.cursor()

# ----------------------------------------------------------
# DROP OLD TABLE (TO AVOID DUPLICATE PRIMARY KEY ERROR)
# ----------------------------------------------------------

cursor.execute("DROP TABLE IF EXISTS students")

# ----------------------------------------------------------
# CREATE TABLE
# ----------------------------------------------------------

cursor.execute("""

CREATE TABLE students (

    id INTEGER PRIMARY KEY,
    name TEXT,
    age INTEGER,
    marks INTEGER

)

""")

print("Table Created Successfully")

# ----------------------------------------------------------
# INSERT DATA
# ----------------------------------------------------------

students_data = [

    (1, 'Abhinav', 20, 85),
    (2, 'Rahul', 21, 78),
    (3, 'Sneha', 19, 92)

]

# Insert multiple records
cursor.executemany(
    "INSERT INTO students VALUES (?, ?, ?, ?)",
    students_data
)

# Save changes permanently
conn.commit()

print("Data Inserted Successfully")

# ----------------------------------------------------------
# DISPLAY ALL RECORDS
# ----------------------------------------------------------

cursor.execute("SELECT * FROM students")

rows = cursor.fetchall()

print("\nStudent Records:\n")

for row in rows:
    print(row)

# ----------------------------------------------------------
# STUDENTS WITH MARKS GREATER THAN 80
# ----------------------------------------------------------

cursor.execute(
    "SELECT * FROM students WHERE marks > 80"
)

print("\nStudents with Marks > 80:\n")

for row in cursor.fetchall():
    print(row)

# ----------------------------------------------------------
# AVERAGE MARKS
# ----------------------------------------------------------

cursor.execute(
    "SELECT AVG(marks) FROM students"
)

print("\nAverage Marks:")

print(cursor.fetchone()[0])

# ----------------------------------------------------------
# UPDATE RECORD
# ----------------------------------------------------------

cursor.execute(
    "UPDATE students SET marks = 88 WHERE id = 2"
)

conn.commit()

print("\nRecord Updated Successfully")

# ----------------------------------------------------------
# DELETE RECORD
# ----------------------------------------------------------

cursor.execute(
    "DELETE FROM students WHERE id = 1"
)

conn.commit()

print("\nRecord Deleted Successfully")

# ----------------------------------------------------------
# FINAL TABLE DATA
# ----------------------------------------------------------

cursor.execute("SELECT * FROM students")

print("\nFinal Table Data:\n")

for row in cursor.fetchall():
    print(row)

# ==========================================================
# EXTRA QUERIES
# ==========================================================

# ----------------------------------------------------------
# MAXIMUM MARKS
# ----------------------------------------------------------

cursor.execute(
    "SELECT MAX(marks) FROM students"
)

print("\nMaximum Marks:")

print(cursor.fetchone()[0])

# ----------------------------------------------------------
# MINIMUM MARKS
# ----------------------------------------------------------

cursor.execute(
    "SELECT MIN(marks) FROM students"
)

print("\nMinimum Marks:")

print(cursor.fetchone()[0])

# ----------------------------------------------------------
# TOTAL STUDENTS
# ----------------------------------------------------------

cursor.execute(
    "SELECT COUNT(*) FROM students"
)

print("\nTotal Students:")

print(cursor.fetchone()[0])

# ----------------------------------------------------------
# SORT STUDENTS BY MARKS
# ----------------------------------------------------------

cursor.execute(
    "SELECT * FROM students ORDER BY marks DESC"
)

print("\nStudents Sorted by Marks:\n")

for row in cursor.fetchall():
    print(row)

# ----------------------------------------------------------
# STUDENTS WITH AGE GREATER THAN 20
# ----------------------------------------------------------

cursor.execute(
    "SELECT * FROM students WHERE age > 20"
)

print("\nStudents with Age > 20:\n")

for row in cursor.fetchall():
    print(row)

# ----------------------------------------------------------
# CLOSE DATABASE CONNECTION
# ----------------------------------------------------------

conn.close()

print("\nDatabase Connection Closed")
