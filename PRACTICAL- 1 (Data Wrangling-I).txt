import pandas as pd
import numpy as np
from sklearn import preprocessing
from sklearn.preprocessing import MinMaxScaler

url = "https://archive.ics.uci.edu/ml/machine-learning-databases/iris/iris.data"

col_names = [
    'Sepal_Length',
    'Sepal_Width',
    'Petal_Length',
    'Petal_Width',
    'Species'
]

df = pd.read_csv(url, names=col_names)

print("\nFirst 5 Rows of Dataset")
print(df.head())

print("\nLast 5 Rows of Dataset")
print(df.tail())

print("\nShape of Dataset (Rows, Columns)")
print(df.shape)

print("\nColumn Names")
print(df.columns)

print("\nIndex Information")
print(df.index)

print("\nData Types")
print(df.dtypes)

print("\nStatistical Summary")
print(df.describe())

print("\nCheck Missing Values")
print(df.isnull())

print("\nMissing Values Column Wise")
print(df.isnull().sum())

print("\nTotal Missing Values in Dataset")
print(df.isnull().sum().sum())

print("\nMissing Values Row Wise")
print(df.isnull().sum(axis=1))

print("\nVariable Description")
print("""
1. Sepal_Length  -> Length of sepal in cm
2. Sepal_Width   -> Width of sepal in cm
3. Petal_Length  -> Length of petal in cm
4. Petal_Width   -> Width of petal in cm
5. Species       -> Type of Iris flower
""")

print("\nBefore Type Conversion")
print(df.dtypes)

df['Petal_Length'] = df['Petal_Length'].astype(int)

print("\nAfter Type Conversion")
print(df.dtypes)

numeric_columns = [
    'Sepal_Length',
    'Sepal_Width',
    'Petal_Length',
    'Petal_Width'
]

scaler = MinMaxScaler()

df_normalized = df.copy()

df_normalized[numeric_columns] = scaler.fit_transform(
    df_normalized[numeric_columns]
)

print("\nNormalized Dataset")
print(df_normalized.head())

print("\nUnique Species Before Encoding")
print(df['Species'].unique())

label_encoder = preprocessing.LabelEncoder()

df['Species'] = label_encoder.fit_transform(df['Species'])

print("\nUnique Species After Encoding")
print(df['Species'].unique())

print("\nDataset After Label Encoding")
print(df.head())

print("\nFirst 3 Rows")
print(df[0:3])

print("\nSelect Specific Columns")
print(df.loc[:, ["Sepal_Length", "Petal_Length"]])

print("\nFirst 5 Rows using iloc")
print(df.iloc[:5, :])

print("\nFirst 3 Columns")
print(df.iloc[:, :3])

print("\nSubset of First 5 Rows and First 3 Columns")
print(df.iloc[:5, :3])

DATA NORAMALIZATION AND MIN MAX

Definition:
Data normalization is the process of scaling data values to a standard range.

Min-Max normalization is a type of normalization that scales values between 0 and 1.

Explanation:
Normalization ensures that all features have the same scale so that no feature dominates others.

Min-Max normalization specifically uses the minimum and maximum values of the data to transform it into a fixed range (usually 0 to 1).

Formula (Min-Max Normalization):

X' = (X - Xmin) / (Xmax - Xmin)

Example:
If data = [10, 20, 30]
→ Normalized (Min-Max) = [0, 0.5, 1]

Viva Line:
Normalization scales data to a common range, and Min-Max normalization is a method that converts values between 0 and 1.

19. What is the limitation of Label Encoding?
Answer:
Label Encoding may create a false order or priority between categories.

Example:
If:

Red = 0

Blue = 1

Green = 2

The model may assume Green > Blue > Red, which may not be true.

So for nominal data, One-Hot Encoding is often preferred.
