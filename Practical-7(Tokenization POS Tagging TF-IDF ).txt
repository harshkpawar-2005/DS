import nltk
import pandas as pd
import re
import math

from nltk.tokenize import sent_tokenize, word_tokenize
from nltk.corpus import stopwords
from nltk.stem import PorterStemmer, WordNetLemmatizer

nltk.download('punkt')
nltk.download('punkt_tab')
nltk.download('stopwords')
nltk.download('wordnet')
nltk.download('averaged_perceptron_tagger')
nltk.download('averaged_perceptron_tagger_eng')

text = """
Natural Language Processing is an important field of Artificial Intelligence.
It helps computers understand human language.
Applications of NLP include chatbots, speech recognition,
machine translation, and sentiment analysis.
"""

print("\n========== ORIGINAL TEXT ==========\n")

print(text)

print("\n========== SENTENCE TOKENIZATION ==========\n")

sentence_tokens = sent_tokenize(text)

print(sentence_tokens)

print("\n========== WORD TOKENIZATION ==========\n")

word_tokens = word_tokenize(text)

print(word_tokens)

print("\n========== STOP WORD REMOVAL ==========\n")

stop_words = set(stopwords.words("english"))

clean_text = re.sub('[^a-zA-Z]', ' ', text)

tokens = word_tokenize(clean_text.lower())

filtered_text = []

for word in tokens:

    if word not in stop_words:

        filtered_text.append(word)

print("Tokenized Words :")

print(tokens)

print("\nFiltered Words :")

print(filtered_text)

print("\n========== STEMMING ==========\n")

ps = PorterStemmer()

example_words = ["wait", "waiting", "waited", "waits"]

for word in example_words:

    root_word = ps.stem(word)

    print(word, "->", root_word)

print("\n========== LEMMATIZATION ==========\n")

lemmatizer = WordNetLemmatizer()

lemma_text = "studies studying cries cry"

tokenization = word_tokenize(lemma_text)

for word in tokenization:

    print("Lemma for", word, "is",
          lemmatizer.lemmatize(word))

print("\n========== POS TAGGING ==========\n")

data = "The pink sweater fit her perfectly"

words = word_tokenize(data)

for word in words:

    print(nltk.pos_tag([word]))

print("\n========== TF-IDF ==========\n")

documentA = "Jupiter is the largest planet"

documentB = "Mars is the fourth planet from the Sun"

bagOfWordsA = documentA.split(' ')

bagOfWordsB = documentB.split(' ')

uniqueWords = set(bagOfWordsA).union(set(bagOfWordsB))

print("Unique Words :")

print(uniqueWords)

numOfWordsA = dict.fromkeys(uniqueWords, 0)

for word in bagOfWordsA:

    numOfWordsA[word] += 1

numOfWordsB = dict.fromkeys(uniqueWords, 0)

for word in bagOfWordsB:

    numOfWordsB[word] += 1

def computeTF(wordDict, bagOfWords):

    tfDict = {}

    bagOfWordsCount = len(bagOfWords)

    for word, count in wordDict.items():

        tfDict[word] = count / float(bagOfWordsCount)

    return tfDict

tfA = computeTF(numOfWordsA, bagOfWordsA)

tfB = computeTF(numOfWordsB, bagOfWordsB)

print("\nTF for Document A :")

print(tfA)

print("\nTF for Document B :")

print(tfB)

def computeIDF(documents):

    N = len(documents)

    idfDict = dict.fromkeys(documents[0].keys(), 0)

    for document in documents:

        for word, val in document.items():

            if val > 0:

                idfDict[word] += 1

    for word, val in idfDict.items():

        idfDict[word] = math.log(N / float(val))

    return idfDict

idfs = computeIDF([numOfWordsA, numOfWordsB])

print("\nIDF Values :")

print(idfs)

def computeTFIDF(tfBagOfWords, idfs):

    tfidf = {}

    for word, val in tfBagOfWords.items():

        tfidf[word] = val * idfs[word]

    return tfidf

tfidfA = computeTFIDF(tfA, idfs)

tfidfB = computeTFIDF(tfB, idfs)

df = pd.DataFrame([tfidfA, tfidfB])

print("\n========== TF-IDF TABLE ==========\n")

print(df)

4. What is Tokenization?
Answer:
Tokenization is the process of breaking text into smaller units called tokens.

Tokens can be:

Words

Sentences

Example:

"I love AI"
becomes:

["I", "love", "AI"]

6. Why is Tokenization important?
Answer:
Tokenization is important because machine learning algorithms cannot directly understand raw text.

It converts text into manageable pieces for further processing.

7. What are Stop Words?
Answer:
Stop words are commonly used words that do not add much meaning to the sentence.

Examples:

is

am

the

are

an

These words are usually removed during preprocessing.



8. Why do we remove stop words?
Answer:
Stop words are removed because:

They increase unnecessary data

They do not contribute much meaning

They reduce processing efficiency

Removing them improves model performance.

9. What is Stemming?
Answer:
Stemming is the process of reducing words to their root form by removing suffixes.

Examples:

waiting → wait

studies → studi

It may not always produce meaningful words.



10. What is Lemmatization?
Answer:
Lemmatization converts words into their meaningful dictionary base form called lemma.

Examples:

studies → study

cries → cry

It gives more meaningful output than stemming.



11. Difference between Stemming and Lemmatization?
Answer:

Stemming	Lemmatization
Removes suffix directly	Uses dictionary and context
Faster	More accurate
May give meaningless words	Gives meaningful root word
Example: studies → studi	Example: studies → study


12. What is POS Tagging?
Answer:
POS Tagging means assigning grammatical tags to words.

POS stands for:

Parts Of Speech

Examples:

NN → Noun

VB → Verb

JJ → Adjective



13. Why is POS Tagging used?
Answer:
POS Tagging helps:

Understand sentence structure

Identify grammatical roles

Improve NLP tasks like sentiment analysis and text understanding

14. What is Bag of Words (BoW)?
Answer:
Bag of Words is a text representation technique.

It converts text into numerical vectors based on word frequency.

It ignores:

Grammar

Word order



15. Why is Bag of Words called “Bag”?
Answer:
It is called “Bag” because:

Word order is ignored

Only word occurrence matters

Sentence structure information is lost.

16. What is Term Frequency (TF)?
Answer:
TF measures how frequently a word appears in a document.

Formula:

TF = Number of times word appears / Total words in document

Higher TF means the word is important in that document.



17. What is Inverse Document Frequency (IDF)?
Answer:
IDF measures how important a word is across all documents.

Words appearing in many documents get lower IDF.

Formula:

IDF = log(N / df)

Where:

N = total documents

df = number of documents containing the word



18. What is TF-IDF?
Answer:
TF-IDF stands for:

Term Frequency × Inverse Document Frequency

Formula:

TF-IDF = TF × IDF

It gives high importance to:

Frequent words in a document

Rare words in the corpus



19. Why is TF-IDF important?
Answer:
TF-IDF helps identify important words in documents.

It is widely used in:

Search engines

Document classification

Recommendation systems

NLP applications

20. What is the disadvantage of TF-IDF?
Answer:
TF-IDF cannot understand semantics or word meaning.

Example:

funny and humorous have similar meaning

TF-IDF treats them as different words

It also becomes expensive for very large vocabularies.
